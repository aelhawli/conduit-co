export default {
  async fetch(request, env) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "*",
    };

    // Handle CORS preflight browser check
    if (request.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: corsHeaders });
    }

    if (request.method === "POST") {
      try {
        const body = await request.json();
        if (!body.pdfBase64) {
          return new Response(JSON.stringify({ error: "No PDF file data provided." }), {
            status: 400,
            headers: { ...corsHeaders, "Content-Type": "application/json" },
          });
        }

        const prompt = `
          You are an expert Australian Construction Estimator for electrical and plumbing contractors.
          Analyze this PDF drawing set and extract:
          1. "summary": A 2-sentence concise summary of the job scope.
          2. "topRisks": Array of 3 high-risk clauses, scope traps, or missing symbols/legends.

          Respond ONLY in this exact JSON format without markdown wrapping:
          {
            "summary": "string",
            "topRisks": [{"title": "string", "description": "string"}]
          }
        `;

        const geminiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${env.GEMINI_API_KEY}`;

        const response = await fetch(geminiUrl, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            contents: [
              {
                parts: [
                  { text: prompt },
                  { inline_data: { mime_type: "application/pdf", data: body.pdfBase64 } }
                ]
              }
            ]
          })
        });

        const geminiData = await response.json();

        if (geminiData.error) {
          return new Response(JSON.stringify({ error: "Gemini Error: " + geminiData.error.message }), {
            status: 400,
            headers: { ...corsHeaders, "Content-Type": "application/json" },
          });
        }

        let rawText = geminiData.candidates[0].content.parts[0].text;
        rawText = rawText.replace(/```json/g, "").replace(/```/g, "").trim();
        const parsed = JSON.parse(rawText);

        return new Response(JSON.stringify({ success: true, audit: parsed }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" },
        });

      } catch (err) {
        return new Response(
          JSON.stringify({ error: "Worker Execution Error: " + err.message }),
          { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }
        );
      }
    }

    return new Response("Conduit Co AI Engine Active", { headers: corsHeaders });
  }
};
